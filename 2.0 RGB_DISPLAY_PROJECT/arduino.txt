void parseStringToArray(const String& data) {
  int index = 0;
  // Serial.println(data);
  // Serial.println("-----------------");
  for (int i = 0; i < ARRAY_SIZE_X; ++i) {
    for (int j = 0; j < ARRAY_SIZE_Y; ++j) {
      for (int k = 0; k < ARRAY_SIZE_Z; ++k) {
        int nextComma = data.indexOf(',', index);
        if (nextComma == -1) {
          nextComma = data.length();
        }
        String valueString = data.substring(index, nextComma);
        dataArray[i][j][k] = valueString.toInt();
        index = nextComma + 1;
      }
      // Serial.print(dataArray[i][j][0]);
      // Serial.print(",");
      // Serial.print(dataArray[i][j][1]);
      // Serial.print(",");
      // Serial.print(dataArray[i][j][2]);
      // Serial.print(",");
    }
  }
}

int rounding(double value) {
  int intValue = (int)value;
  double fractionalPart = value - intValue;

  if (fractionalPart >= 0.5) {
    return intValue + 1; // Round up
  } else {
    return intValue; // Round down
  }
}
